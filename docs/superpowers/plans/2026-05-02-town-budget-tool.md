> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship `/town-budget.html` — an interactive, hierarchical line-item explorer for the FY27 town budget with tree-table UI, deep filters, sortable columns, group-by toggles, shareable URL state, and ACFR history at department level back to FY15.

**Architecture:** Single-file Jekyll page (frontmatter + inline `<style>` + inline `<script>`) backed by a committed JSON file generated by a Python parser. JSON produced from the FY27 Proposed Budget book text dump and the school dept's FY27 budget packet, joined to ACFR department actuals via a hand-curated alias map. Vanilla JS only. Playwright tests use the existing pass/fail counter pattern in `tests/smoke-test.mjs`. Python parser tested with pytest (new — first pytest in this repo).

**Tech Stack:** Jekyll 3.10, vanilla JS (no framework, no chart lib), Python 3 with stdlib + PyYAML, pytest, Playwright (Chromium), inline SVG sparklines.

```python
#!/usr/bin/env python3
"""Parse the FY27 Proposed Budget book into a structured row list.

Source: data/FY27_Proposed_Budget_No_Override.txt (whitespace-delimited
text dump of the FY27 budget book PDF).

Output: data/town_budget_FY27.json + data/town_budget_FY27_lookup.json

Usage:
  python3 data/build_town_budget_data.py
"""
from __future__ import annotations
import json
import re
from pathlib import Path
from typing import Iterable, Optional

ROOT = Path(__file__).resolve().parent.parent
DATA = ROOT / "data"

# Pattern matches a line like:
#   "TOTAL BUDGETS    115,368,206   111,991,311   119,479,480   122,762,030   3,282,550   2.75%"
# Six numeric columns, last is a percentage.
_NUM = r"[\d,]+|-"
_PCT = r"-?[\d.]+%"
_GRAND_TOTAL_RE = re.compile(
    rf"^\s*TOTAL BUDGETS\s+({_NUM})\s+({_NUM})\s+({_NUM})\s+({_NUM})\s+(\(?{_NUM}\)?)\s+({_PCT})",
    re.MULTILINE,
)


def _parse_int(s: str) -> Optional[int]:
    s = s.strip().replace(",", "")
    if s in {"", "-"}:
        return 0
    if s.startswith("(") and s.endswith(")"):
        return -int(s[1:-1])
    return int(s)


def _parse_pct(s: str) -> Optional[float]:
    s = s.strip().rstrip("%")
    if s in {"", "-"}:
        return 0.0
    return float(s) / 100.0


def parse_budget_book(text: str) -> list[dict]:
    """Parse the FY27 budget book text into a flat list of rows.

    Rows have fields: id, level, parent_id, function, department,
    description, spend_type, fy25_actual, fy25_budget, fy26_budget,
    fy27_proposed, change_dollars, change_pct.
    """
    rows: list[dict] = []

    m = _GRAND_TOTAL_RE.search(text)
    if m:
        rows.append({
            "id": "total_budgets",
            "level": "grand_total",
            "parent_id": None,
            "function": None,
            "department": None,
            "description": "TOTAL BUDGETS",
            "spend_type": None,
            "fy25_budget": _parse_int(m.group(1)),
            "fy25_actual": _parse_int(m.group(2)),
            "fy26_budget": _parse_int(m.group(3)),
            "fy27_proposed": _parse_int(m.group(4)),
            "change_dollars": _parse_int(m.group(5)),
            "change_pct": _parse_pct(m.group(6)),
        })

    return rows


if __name__ == "__main__":
    text = (DATA / "FY27_Proposed_Budget_No_Override.txt").read_text()
    rows = parse_budget_book(text)
    print(json.dumps(rows[:5], indent=2))
```

Replace the body of `parse_budget_book()` so it does a line-by-line pass that tracks current function (advancing each time a VOTE TOTAL is hit) and current department (advancing on each department header), and aggregates line items into department subtotals as they're encountered:

```python
def parse_budget_book(text: str) -> list[dict]:
    rows: list[dict] = []

    # Pass 1: extract grand totals + function totals (already implemented above).
    m = _GENERAL_FUND_TOTAL_RE.search(text)
    if m:
        rows.append(_make_grand_total_row("total_general_fund",
                                          "TOTAL GENERAL FUND ACCOUNTS", m))
    for m in _VOTE_TOTAL_RE.finditer(text):
        descr = m.group(1).strip()
        slug = _FUNCTION_DESC_TO_SLUG.get(descr)
        if slug:
            rows.append(_make_function_row(slug, descr, m))
    m = _GRAND_TOTAL_RE.search(text)
    if m:
        rows.append(_make_grand_total_row("total_budgets", "TOTAL BUDGETS", m))

    # Pass 2: line items and department headers.
    current_function_idx = 0
    current_dept_slug: Optional[str] = None
    current_dept_descr: Optional[str] = None
    dept_subtotals: dict[str, dict] = {}  # slug → row

    lines = text.splitlines()
    for raw in lines:
        if "VOTE TOTAL" in raw and current_function_idx < len(_FUNCTION_ORDER):
            current_function_idx += 1
            current_dept_slug = None
            continue

        line_m = _LINE_ITEM_RE.match(raw)
        if line_m:
            num, descr = line_m.group(1), line_m.group(2).strip()
            fy25_b = _parse_int(line_m.group(3))
            fy25_a = _parse_int(line_m.group(4))
            fy26_b = _parse_int(line_m.group(5))
            fy27_p = _parse_int(line_m.group(6))
            chg_d = _parse_int(line_m.group(7))
            chg_p = _parse_pct(line_m.group(8))
            fn_slug = (_FUNCTION_ORDER[current_function_idx]
                       if current_function_idx < len(_FUNCTION_ORDER) else None)
            line_id = f"line_{num}"
            rows.append({
                "id": line_id,
                "level": "line",
                "parent_id": current_dept_slug,
                "function": fn_slug,
                "department": current_dept_slug,
                "description": descr,
                "spend_type": classify_spend_type(descr),
                "fy25_budget": fy25_b,
                "fy25_actual": fy25_a,
                "fy26_budget": fy26_b,
                "fy27_proposed": fy27_p,
                "change_dollars": chg_d,
                "change_pct": chg_p,
                "source_ref": {
                    "doc": "fy27_budget_book",
                    "line_item": int(num),
                },
            })
            # Aggregate dept subtotal as we go.
            if current_dept_slug:
                d = dept_subtotals.setdefault(current_dept_slug, {
                    "id": current_dept_slug,
                    "level": "department",
                    "parent_id": fn_slug,
                    "function": fn_slug,
                    "department": current_dept_slug,
                    "description": current_dept_descr,
                    "spend_type": None,
                    "fy25_budget": 0, "fy25_actual": 0,
                    "fy26_budget": 0, "fy27_proposed": 0,
                    "change_dollars": 0,
                })
                d["fy25_budget"] += fy25_b
                d["fy25_actual"] += fy25_a
                d["fy26_budget"] += fy26_b
                d["fy27_proposed"] += fy27_p
                d["change_dollars"] += chg_d
            continue

        dept_m = _DEPT_HEADER_RE.match(raw)
        if dept_m:
            dept_descr = dept_m.group(1).strip()
            dept_slug = _DEPT_DISPLAY_TO_SLUG.get(dept_descr)
            if dept_slug is None:
                # Auto-slug if unknown — this is a soft warning case; tests
                # ensure all FY27 departments are in the map.
                dept_slug = re.sub(r"[^a-z0-9]+", "_", dept_descr.lower()).strip("_")
            current_dept_slug = dept_slug
            current_dept_descr = dept_descr

    # Compute change_pct for dept subtotals and append.
    for d in dept_subtotals.values():
        prior = d["fy26_budget"]
        d["change_pct"] = (d["change_dollars"] / prior) if prior else 0.0
        rows.append(d)

    return rows

def classify_spend_type(descr: str) -> str:
    """Bucket a line-item description into a spend type.

    Buckets: salaries | expense | officials_expense | benefits | debt
    | transfer | utility | reserve | other.
    """
    d = descr.lower()
    if "salaries" in d or "salary reserve" in d:
        return "salaries"
    if "officials expense" in d:
        return "officials_expense"
    if "expense" in d:
        return "expense"
    if "benefits" in d or "retirement" in d or "insurance" in d \
       or "medicare" in d or "unemployment" in d \
       or "post employment" in d or "workers compensation" in d \
       or "flex spending" in d:
        return "benefits"
    if "maturing debt" in d or "interest" in d or "bonds" in d:
        return "debt"
    if "transfer" in d or "stabilization" in d:
        return "transfer"
    if "lighting" in d or "utility reserve" in d or "energy reserve" in d:
        return "utility"
    if "reserves" in d or "reserve fund" in d:
        return "reserve"
    return "other"
```

```python
def load_dept_aliases() -> dict[str, list[str]]:
    """Return slug → list of ACFR display-name aliases."""
    p = DATA / "dept_aliases.yaml"
    raw = yaml.safe_load(p.read_text())
    return {slug: cfg.get("acfr_names", []) for slug, cfg in raw.items()}


def load_acfr_history() -> dict[str, dict[str, int]]:
    """Return acfr_name → {fyXX_actual: int} for years FY15-FY24.

    Reads data/general_fund_budgetary_FY15-24.csv.
    Schema (verify by inspecting the file): first column is dept/function
    name, subsequent columns are FY15..FY24 actuals.
    """
    p = DATA / "general_fund_budgetary_FY15-24.csv"
    history: dict[str, dict[str, int]] = {}
    with p.open() as f:
        reader = csv.DictReader(f)
        for row in reader:
            name = (row.get("dept_or_function")
                    or row.get("department")
                    or row.get("name") or "").strip()
            if not name:
                continue
            history[name] = {}
            for col in reader.fieldnames:
                if col.lower().startswith("fy") and col.lower().endswith("_actual"):
                    val = row[col].replace(",", "").strip()
                    if val and val != "-":
                        history[name][col.lower()] = int(float(val))
    return history


def attach_history(rows: list[dict]) -> None:
    """Attach 10-yr `history` and `cagr_10yr` to department-level rows."""
    aliases = load_dept_aliases()
    history_by_name = load_acfr_history()
    for r in rows:
        if r["level"] != "department":
            continue
        slug = r["id"]
        candidates = aliases.get(slug, [])
        for name in candidates:
            if name in history_by_name:
                r["history"] = history_by_name[name]
                # 10-yr CAGR from FY15 → FY24 actuals.
                start = r["history"].get("fy15_actual")
                end = r["history"].get("fy24_actual")
                if start and end and start > 0:
                    r["cagr_10yr"] = (end / start) ** (1 / 9) - 1
                break
```

```python
def parse_school_packet(text: str) -> list[dict]:
    """Parse the school dept's FY27 budget packet for cost-center totals.

    The packet's exact layout varies — implementation reads the
    text dump and extracts lines matching `<cost-center-name>  $X,XXX,XXX`
    or similar tabular forms. If the packet format changes meaningfully,
    this function returns [] and the tool falls back to a single $47.6M
    schools row.
    """
    rows: list[dict] = []
    # Heuristic: match lines like "Brown School  4,374,821" (commas, no $).
    pat = re.compile(
        r"^[\s ]*([A-Z][A-Za-z &\-/.']+?)[\s ]+\$?([\d,]+(?:\.\d+)?)\s*$",
        re.MULTILINE,
    )
    found_amounts: dict[str, int] = {}
    for m in pat.finditer(text):
        descr = m.group(1).strip()
        amt = _parse_int(m.group(2))
        slug = _SCHOOL_DESC_TO_SLUG.get(descr)
        if slug and amt and amt > 100_000:  # skip small numbers
            # Take the first plausible occurrence per cost center.
            found_amounts.setdefault(slug, amt)
            display = descr
            if slug not in {r["id"] for r in rows}:
                rows.append({
                    "id": slug,
                    "level": "department",
                    "parent_id": "schools",
                    "function": "schools",
                    "department": slug,
                    "description": display,
                    "spend_type": None,
                    "fy27_proposed": amt,
                    # Other years/changes are unknown without further parsing —
                    # set to None so the UI shows "—" for them.
                    "fy26_budget": None,
                    "fy25_budget": None,
                    "fy25_actual": None,
                    "change_dollars": None,
                    "change_pct": None,
                    "source_ref": {"doc": "fy27_school_packet"},
                })
    return rows
```

```javascript
function buildVisibleRows() {
  var byParent = {};
  data.rows.forEach(r => {
    if (r.parent_id) {
      (byParent[r.parent_id] = byParent[r.parent_id] || []).push(r);
    }
```

```javascript
function renderSparkline(history) {
  if (!history) return '';
  // Pull sequential FY15..FY24 actuals; nulls become gaps.
  var years = ['fy15_actual','fy16_actual','fy17_actual','fy18_actual',
               'fy19_actual','fy20_actual','fy21_actual','fy22_actual',
               'fy23_actual','fy24_actual'];
  var values = years.map(y => history[y] || null).filter(v => v != null);
  if (values.length < 2) return '';
  var min = Math.min(...values), max = Math.max(...values);
  var w = 80, h = 18, pad = 1;
  var span = max - min || 1;
  var pts = values.map((v, i) =>
    [pad + (i * (w - 2 * pad)) / (values.length - 1),
     h - pad - ((v - min) / span) * (h - 2 * pad)]

  var d = pts.map((p, i) => (i === 0 ? 'M' : 'L') + p[0].toFixed(1) + ' ' + p[1].toFixed(1)).join('');

           '<path d="' + d + '"/>' +
```

```javascript
function buildVisibleRows() {
  if (state.groupBy === 'spend_type') {
    // Build pseudo-groups: each spend_type is a synthetic top-level row,
    // children are the line items with that spend_type.
    var groups = {};
    var groupChildren = {};
    data.rows.filter(r => r.level === 'line').forEach(r => {
      var t = r.spend_type || 'other';
      var groupId = 'spend_' + t;
      if (!groups[groupId]) {
        groups[groupId] = {
          id: groupId, level: 'function',
          parent_id: null, function: groupId, department: null,
          description: t.charAt(0).toUpperCase() + t.slice(1).replace('_', ' '),
          spend_type: null,
          fy25_budget: 0, fy25_actual: 0, fy26_budget: 0,
          fy27_proposed: 0, change_dollars: 0, change_pct: 0,
        };
        groupChildren[groupId] = [];
      }
      var g = groups[groupId];
      ['fy25_budget','fy25_actual','fy26_budget','fy27_proposed','change_dollars']
        .forEach(k => { g[k] += r[k] || 0; });
      // Re-parent the line in this view so it descends under the group.
      var rebased = Object.assign({}, r, { parent_id: groupId });
      groupChildren[groupId].push(rebased);
    });
    Object.values(groups).forEach(g => {
      g.change_pct = g.fy26_budget ? g.change_dollars / g.fy26_budget : 0;
    });

    // Walk groups, expand children if state.expanded.has(groupId).
    var visible = [];
    var orderedGroups = Object.values(groups).sort((a, b) => {
      var av = a[state.sortKey] || 0, bv = b[state.sortKey] || 0;
      return (state.sortDir === 'asc' ? 1 : -1) * (av < bv ? -1 : av > bv ? 1 : 0);
    });
    orderedGroups.forEach(g => {
      if (!rowMatchesFilters(g) && !state.filters.searchQuery) return;
      visible.push(g);
      if (state.expanded.has(g.id)) {
        var kids = groupChildren[g.id].slice().sort((a, b) => {
          var av = a[state.sortKey] || 0, bv = b[state.sortKey] || 0;
          return (state.sortDir === 'asc' ? 1 : -1) * (av < bv ? -1 : av > bv ? 1 : 0);
        });
        kids.forEach(k => {
          if (rowMatchesFilters(k)) visible.push(k);
        });
      }
    });
    return visible;
  }

  // Default: function grouping. (This is the body that existed before
  // the spend_type branch was added — reproduced in full here.)
  var byParent = {};
  data.rows.forEach(r => {
    if (r.parent_id) {
      (byParent[r.parent_id] = byParent[r.parent_id] || []).push(r);
    }
  });
  var ORDER = [
    'general_government', 'public_safety', 'public_works', 'human_services',
    'schools', 'culture_recreation', 'other_general_government',
    'total_general_fund',
    'sewer_enterprise', 'water_enterprise', 'harbor_enterprise',
    'total_budgets'

      if (!state.filters.includeEnterprise &&
          ['sewer_enterprise','water_enterprise','harbor_enterprise'].indexOf(r.id) >= 0) {
        return false;
      }
      if (r.function && state.filters.functions.size > 0) {
        return state.filters.functions.has(r.function);
      }
      return true;
    });

  var forced = ancestorsOfMatching();
  var visible = [];
  function walk(r) {
    if (!rowMatchesFilters(r) && r.level === 'line') return;
    if (r.level !== 'line' && !rowMatchesFilters(r)) {
      // Non-line row failing filter still shown if it has matching descendants
      // because of `forced`. If not forced, skip.
      if (!forced.has(r.id)) return;
    }
    visible.push(r);
    var open = state.expanded.has(r.id) || forced.has(r.id);
    if (open) {
      var children = (byParent[r.id] || []).slice();
      children.sort((a, b) => {
```

```javascript
function parseURLState() {
  var p = new URLSearchParams(location.search);
  var s = makeDefaultState();
  if (p.has('fn')) s.filters.functions = new Set(p.get('fn').split(','));
  if (p.has('type')) s.filters.spendTypes = new Set(p.get('type').split(','));
  if (p.has('dirfilter')) s.filters.directions = new Set(p.get('dirfilter').split(','));
  if (p.has('pct_min')) s.filters.pctMin = parseFloat(p.get('pct_min'));
  if (p.has('dollar_min')) s.filters.dollarMin = parseFloat(p.get('dollar_min'));
  if (p.get('gf') === 'all') s.filters.includeEnterprise = true;
  if (p.has('q')) s.filters.searchQuery = p.get('q').toLowerCase();
  if (p.has('dept')) s.filters.deptQuery = p.get('dept').toLowerCase();
  if (p.has('sort')) s.sortKey = p.get('sort');
  if (p.has('dir')) s.sortDir = p.get('dir');
  if (p.has('group')) s.groupBy = p.get('group');
  if (p.has('cols')) s.columns = new Set(p.get('cols').split(','));
  if (p.has('expand')) p.get('expand').split(',').forEach(id => s.expanded.add(id));
  return s;
}

function serializeURLState() {
  var d = makeDefaultState();
  var p = new URLSearchParams();
  if (state.filters.functions.size !== d.filters.functions.size) {
    p.set('fn', [...state.filters.functions].join(','));
  }
  if (state.filters.spendTypes.size !== d.filters.spendTypes.size) {
    p.set('type', [...state.filters.spendTypes].join(','));
  }
  if (state.filters.directions.size !== d.filters.directions.size) {
    p.set('dirfilter', [...state.filters.directions].join(','));
  }
  if (state.filters.pctMin != null) p.set('pct_min', state.filters.pctMin);
  if (state.filters.dollarMin != null) p.set('dollar_min', state.filters.dollarMin);
  if (state.filters.includeEnterprise) p.set('gf', 'all');
  if (state.filters.searchQuery) p.set('q', state.filters.searchQuery);
  if (state.filters.deptQuery) p.set('dept', state.filters.deptQuery);
  if (state.sortKey !== d.sortKey) p.set('sort', state.sortKey);
  if (state.sortDir !== d.sortDir) p.set('dir', state.sortDir);
  if (state.groupBy !== d.groupBy) p.set('group', state.groupBy);
  if (state.expanded.size > 0) p.set('expand', [...state.expanded].join(','));
  var qs = p.toString();
  history.replaceState(null, '', location.pathname + (qs ? '?' + qs : ''));
}
```
